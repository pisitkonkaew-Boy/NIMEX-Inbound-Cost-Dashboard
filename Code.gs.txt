/**
 * NIMEX Inbound Cost Dashboard — Google Apps Script (Web App)
 * ใช้ Google Sheet เดิมได้เลย: ชีต "data" 14 คอลัมน์ลำดับเดิม ข้อมูลเดิมไม่หาย
 *
 * ===================== วิธีติดตั้ง / อัปเดต =====================
 * 1) เปิด Google Sheet เดิม -> ส่วนขยาย (Extensions) -> Apps Script
 * 2) ลบโค้ดเดิมทั้งหมด แล้ววางโค้ดทั้งไฟล์นี้ลงไป กดบันทึก
 * 3) ตั้งรหัสผู้ดูแล: ไอคอนเฟือง "การตั้งค่าโปรเจ็กต์" (Project Settings)
 *      -> Script Properties -> เพิ่มพร็อพเพอร์ตี้  ชื่อ ADMIN_KEY  ค่า = รหัสที่ต้องการ
 *    (รหัสไม่อยู่ในไฟล์นี้และไม่อยู่ในหน้าเว็บ ตรวจที่นี่ที่เดียว)
 * 4) Deploy -> Manage deployments -> ดินสอแก้ไข -> Version: New version -> Deploy
 *    (ใช้ URL /exec เดิม ไม่ต้องแก้ config.js)  ถ้ายังไม่เคย Deploy ให้ใช้ New deployment
 *      Execute as: Me   ·   Who has access: Anyone
 * ===================================================================
 *
 * GET  <URL>                      -> ข้อมูลสาธารณะ {ok, rows:[[14 ค่า]], settings, updated}
 * POST text/plain {a, key, ...}   -> งานผู้ดูแล (ตรวจ ADMIN_KEY ทุกครั้ง)
 *      a = 'auth' | 'upsert' (rows) | 'settings' (settings) | 'deleteMonth' (month 'MM/YYYY') | 'stats'
 */

var SHEET_NAME = 'data';
var FIELDS = ['i', 'dc', 'dn', 'dt', 'pl', 'g', 'rc', 'rd', 'rty', 'bl', 'bx', 'rv', 'co', 'pt'];
var HEADERS_TH = ['เลขที่ใบรายการ', 'DC', 'ชื่อ DC', 'วันที่ (พ.ศ.)', 'ทะเบียนรถ', 'กลุ่ม (AB/CC)',
  'รหัสเส้นทาง', 'เส้นทาง', 'ประเภทอัตรา', 'บิล', 'กล่อง', 'รายได้', 'ค่าจ้าง (จริง)', 'จุดเข้ารับ'];
var TEXT_COLS = 9;
var NCOLS = FIELDS.length;
var MAX_ROWS_PER_POST = 5000;
var DEFAULT_SETTINGS = { kpi: 10, warnBuffer: 1.5 };
var CK = 'nim-inbound';

function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }

function dataSheet_() {
  var sh = ss_().getSheetByName(SHEET_NAME);
  if (!sh) sh = ss_().insertSheet(SHEET_NAME);
  if (sh.getLastRow() === 0) { sh.getRange(1, 1, 1, NCOLS).setValues([HEADERS_TH]); sh.setFrozenRows(1); }
  return sh;
}
function tab_(name, head) {
  var sh = ss_().getSheetByName(name);
  if (!sh) { sh = ss_().insertSheet(name); sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold'); sh.setFrozenRows(1); }
  return sh;
}

function dateText_(v) {
  if (v instanceof Date) {
    var tz = Session.getScriptTimeZone(), y = Number(Utilities.formatDate(v, tz, 'yyyy'));
    return Utilities.formatDate(v, tz, 'dd/MM/') + (y < 2400 ? y + 543 : y);
  }
  return String(v == null ? '' : v).trim();
}
function cleanRow_(r) {
  var out = [];
  for (var c = 0; c < NCOLS; c++) {
    var v = r[c];
    if (c === 3) out.push(dateText_(v));
    else if (c < TEXT_COLS) out.push(String(v == null ? '' : v).trim());
    else out.push(Number(v) || 0);
  }
  return out;
}
function objToRow_(o) { var a = []; for (var c = 0; c < NCOLS; c++) a.push(o[FIELDS[c]]); return cleanRow_(a); }

function readRows_() {
  var sh = dataSheet_(), n = sh.getLastRow();
  if (n < 2) return [];
  return sh.getRange(2, 1, n - 1, NCOLS).getValues().map(cleanRow_).filter(function (r) { return r[0]; });
}
function writeRows_(rows) {
  var sh = dataSheet_(), n = sh.getLastRow();
  if (n > 1) sh.getRange(2, 1, n - 1, NCOLS).clearContent();
  if (!rows.length) return;
  sh.getRange(2, 1, Math.max(rows.length, 1), TEXT_COLS).setNumberFormat('@');
  sh.getRange(2, 1, rows.length, NCOLS).setValues(rows);
}

function readSettings_() {
  var sh = tab_('settings', ['key', 'value']), n = sh.getLastRow(), s = {};
  for (var k in DEFAULT_SETTINGS) s[k] = DEFAULT_SETTINGS[k];
  if (n > 1) sh.getRange(2, 1, n - 1, 2).getValues().forEach(function (r) { var v = Number(r[1]); if (r[0] && isFinite(v)) s[String(r[0])] = v; });
  return s;
}
function writeSettings_(s) {
  var sh = tab_('settings', ['key', 'value']), n = sh.getLastRow();
  if (n > 1) sh.getRange(2, 1, n - 1, 2).clearContent();
  var rows = Object.keys(s).map(function (k) { return [k, s[k]]; });
  if (rows.length) sh.getRange(2, 1, rows.length, 2).setValues(rows);
}
function log_(a, d) { tab_('log', ['time', 'action', 'detail']).appendRow([new Date(), a, d]); }

/* ---------- cache (gzip, split into 90k chunks) ---------- */
function cacheGet_() {
  var c = CacheService.getScriptCache(), n = c.get(CK + '-n'); if (!n) return null;
  var keys = []; for (var i = 0; i < +n; i++) keys.push(CK + '-' + i);
  var all = c.getAll(keys), b = '';
  for (i = 0; i < keys.length; i++) { if (!all[keys[i]]) return null; b += all[keys[i]]; }
  return Utilities.ungzip(Utilities.newBlob(Utilities.base64Decode(b), 'application/x-gzip')).getDataAsString('UTF-8');
}
function cachePut_(s) {
  var b = Utilities.base64Encode(Utilities.gzip(Utilities.newBlob(s, 'text/plain')).getBytes()), o = {}, n = 0;
  for (var i = 0; i < b.length; i += 90000) o[CK + '-' + (n++)] = b.substr(i, 90000);
  if (n <= 90) { o[CK + '-n'] = String(n); CacheService.getScriptCache().putAll(o, 21600); }
}
function cacheClear_() { CacheService.getScriptCache().remove(CK + '-n'); }

function publicJson_() {
  return JSON.stringify({ ok: true, v: 2, cols: FIELDS, rows: readRows_(), settings: readSettings_(), updated: new Date().toISOString() });
}

function doGet() {
  var s = null;
  try { s = cacheGet_(); } catch (e) {}
  if (!s) { s = publicJson_(); try { cachePut_(s); } catch (e) {} }
  return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var b;
  try { b = JSON.parse(e.postData.contents); } catch (x) { return json_({ ok: false, error: 'bad_request' }); }
  var key = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY');
  if (!key) return json_({ ok: false, error: 'admin_key_not_set' });
  if (!b.key || b.key !== key) { Utilities.sleep(1200); return json_({ ok: false, error: 'denied' }); }
  if (b.a === 'auth') return json_({ ok: true });

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) return json_({ ok: false, error: 'busy' });
  try {
    if (b.a === 'upsert') {
      var rows = b.rows || [];
      if (!Array.isArray(rows)) return json_({ ok: false, error: 'bad_rows' });
      if (rows.length > MAX_ROWS_PER_POST) return json_({ ok: false, error: 'too_many_rows' });
      var all = readRows_(), index = {}, added = 0, updated = 0;
      all.forEach(function (r, i) { index[r[0]] = i; });
      rows.forEach(function (o) {
        var r = Array.isArray(o) ? cleanRow_(o) : objToRow_(o);
        if (!r[0]) return;
        if (Object.prototype.hasOwnProperty.call(index, r[0])) { all[index[r[0]]] = r; updated++; }   // ซ้ำ -> ยึดค่าใหม่
        else { index[r[0]] = all.length; all.push(r); added++; }
      });
      writeRows_(all);
      log_('upsert', 'added ' + added + ', updated ' + updated);
      cacheClear_();
      return json_({ ok: true, added: added, updated: updated, total: all.length });
    }
    if (b.a === 'settings') {
      var s = readSettings_(), inp = b.settings || {};
      ['kpi', 'warnBuffer'].forEach(function (k) { var v = Number(inp[k]); if (isFinite(v) && v >= 0) s[k] = v; });
      writeSettings_(s); log_('settings', JSON.stringify(s)); cacheClear_();
      return json_({ ok: true, settings: s });
    }
    if (b.a === 'deleteMonth') {
      var m = String(b.month || '');
      if (!/^\d{2}\/\d{4}$/.test(m)) return json_({ ok: false, error: 'bad_month' });
      var before = readRows_(), keep = before.filter(function (r) { return String(r[3]).slice(-7) !== m; });
      writeRows_(keep); log_('deleteMonth', m + ' removed ' + (before.length - keep.length)); cacheClear_();
      return json_({ ok: true, removed: before.length - keep.length, total: keep.length });
    }
    if (b.a === 'stats') {
      var lg = tab_('log', ['time', 'action', 'detail']), n = lg.getLastRow();
      var last = n > 1 ? lg.getRange(Math.max(2, n - 19), 1, Math.min(20, n - 1), 3).getValues().reverse()
        .map(function (r) { return [Utilities.formatDate(new Date(r[0]), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm'), String(r[1]), String(r[2])]; }) : [];
      return json_({ ok: true, log: last });
    }
    return json_({ ok: false, error: 'unknown' });
  } finally { lock.releaseLock(); }
}
